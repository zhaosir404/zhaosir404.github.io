fcntl就是一个对文件描述符进行控制的库
#include <fcntl.h>
int fcntl(int fd, int cmd, ...)
cmd常用的命令有F_GETFL  F_SETFL(FL可以理解成 flags)
int flags = fcntl(fd, F_GETFL, 0);  // 这里的第三个参数可写可不写
fcntl(fd, F_SETFL, flags | O_NONBLOCK);
一般用get再set只改变非阻塞这个状态位

常用的封装
int set_nonblock(int fd) {
    int flags = fcntl(fd, F_GETFL, 0);
    if (flags < 0) {
        return -1;
    }

    if (fcntl(fd, F_SETFL, flags | O_NONBLOCK) < 0) {
        return -1;
    }

    return 0;
}
