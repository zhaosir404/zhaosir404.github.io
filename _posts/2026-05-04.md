3个函数：
epoll_create1 epoll_ctl epoll_wait
1个结构体
epoll_event 

基本流程
1. socket
2. bind
3. listen
4. epoll_create1
5. 把listen_fd用epoll_ctl(...ADD...) 加进去
6. epoll_wait死循环等事件
7. 事件可分为三种 新链接 数据传输 关闭链接
8. 如果是listen_fd就绪，说明有新连接，accept并把新conn_fd也加进epoll
9. 如果是某个conn_fd就绪，recv/send
10. 客户端断开时，把这个fd从epoll删掉并close

代码
int client_action(){
    int client_fd = socket(AF_INET, SOCK_STREAM, 0);
    if (client_fd < 0)
    {
        return -1;
    }
    sockaddr_in addr{};
    addr.sin_family = AF_INET;
    addr.sin_port = htons(8080);
    inet_pton(AF_INET, "192.168.0.1", &addr.sin_addr);
    if(connect(client_fd, (sockaddr*)&addr, sizeof(addr)) < 0){
        close(client_fd);
        return -1;
    }
    std::string s = "666";
    if (send(client_fd, s.data(), s.size(), 0) != s.size())
    {
        close(client_fd);
        return -1;
    }
    close(client_fd);
    return 0;
}

int server_action() {
    int server_fd = socket(AF_INET, SOCK_STREAM | SOCK_NONBLOCK, 0);
    if (server_fd < 0) {
        return -1;
    }

    sockaddr_in addr{};
    addr.sin_family = AF_INET;
    addr.sin_port = htons(8080);
    addr.sin_addr.s_addr = htonl(INADDR_ANY);

    if (bind(server_fd, reinterpret_cast<sockaddr*>(&addr), sizeof(addr)) < 0) {
        close(server_fd);
        return -1;
    }
    if (listen(server_fd, 5) < 0) {
        close(server_fd);
        return -1;
    }

    int epoll_fd = epoll_create1(0);
    if (epoll_fd < 0) {
        close(server_fd);
        return -1;
    }

    epoll_event ev{};
    ev.events = EPOLLIN | EPOLLET | EPOLLRDHUP;
    ev.data.fd = server_fd;
    if (epoll_ctl(epoll_fd, EPOLL_CTL_ADD, server_fd, &ev) < 0) {
        close(epoll_fd);
        close(server_fd);
        return -1;
    }

    epoll_event events[1024];

    while (true) {
        int events_num = epoll_wait(epoll_fd, events, 1024, -1);
        if (events_num < 0) {
            if (errno == EINTR) {
                continue;
            }
            break;
        }

        for (int i = 0; i < events_num; ++i) {
            int fd = events[i].data.fd;
            uint32_t revents = events[i].events;

            if (fd == server_fd) {
                while (true) {
                    sockaddr_in conn_addr{};
                    socklen_t conn_len = sizeof(conn_addr);
                    int conn_fd = accept4(
                        server_fd,
                        reinterpret_cast<sockaddr*>(&conn_addr),
                        &conn_len,
                        SOCK_NONBLOCK
                    );

                    if (conn_fd < 0) {
                        if (errno == EAGAIN || errno == EWOULDBLOCK) {
                            break;
                        }
                        if (errno == EINTR) {
                            continue;
                        }
                        break;
                    }

                    epoll_event client_ev{};
                    client_ev.events = EPOLLIN | EPOLLET | EPOLLRDHUP;
                    client_ev.data.fd = conn_fd;
                    if (epoll_ctl(epoll_fd, EPOLL_CTL_ADD, conn_fd, &client_ev) < 0) {
                        close(conn_fd);
                    }
                }
                continue;
            }

            if (revents & (EPOLLERR | EPOLLHUP | EPOLLRDHUP)) {
                epoll_ctl(epoll_fd, EPOLL_CTL_DEL, fd, nullptr);
                close(fd);
                continue;
            }

            if (revents & EPOLLIN) {
                while (true) {
                    char buf[1024];
                    ssize_t n = recv(fd, buf, sizeof(buf), 0);

                    if (n > 0) {
                        // 这里是“收到数据”
                        // buf[0..n-1] 有效
                    } else if (n == 0) {
                        // 这里是“对端正常关闭”
                        epoll_ctl(epoll_fd, EPOLL_CTL_DEL, fd, nullptr);
                        close(fd);
                        break;
                    } else {
                        if (errno == EINTR) {
                            continue;
                        }
                        if (errno == EAGAIN || errno == EWOULDBLOCK) {
                            // 这里不是关闭，只是这一轮数据读完了
                            break;
                        close(fd);
                        break;
                    }
                }
            }
        }
    }

    close(epoll_fd);
    close(server_fd);
    return -1;
}
