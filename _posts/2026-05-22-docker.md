docker pull 镜像名  # 拉取镜像
docker ps  # 查看在跑的进程
docker run -d 镜像名  # 运行容器 镜像名前面加参数选项 后面加容器内部命令
-v 宿主机目录：容器目录  # 挂载卷，即宿主机容器使用共享空间，但容器删了实际数据还在
-p 宿主机端口:容器端口
--name 容器名
-e ***=***
-p端口映射 -d后台运行 -e设置环境变量
docker rm 容器id  # 删除容器
docker rm 镜像id  # 删除镜像
