# 客户端相关
int client_action() {
    int fd = socket(AF_INET, SOCK_STREAM, 0);
    if (fd < 0) {
        return -1;
    }

    sockaddr_in addr{};
    addr.sin_family = AF_INET;
    addr.sin_port = htons(8080);
    if (inet_pton(AF_INET, "192.168.0.1", &addr.sin_addr) != 1) {
        close(fd);
        return -1;
    }

    if (connect(fd, reinterpret_cast<sockaddr*>(&addr), sizeof(addr)) < 0) {
        close(fd);
        return -1;
    }

    std::string buf = "666";
    ssize_t send_len = send(fd, buf.data(), buf.size(), 0);
    if (send_len < 0 || static_cast<size_t>(send_len) != buf.size()) {
        close(fd);
        return -1;
    }

    close(fd);
    return 0;
}

# 服务端相关
int server_action() {
    int listen_fd = socket(AF_INET, SOCK_STREAM, 0);
    if (listen_fd < 0) {
        return -1;
    }

    sockaddr_in addr{};
    addr.sin_family = AF_INET;
    addr.sin_port = htons(8080);
    addr.sin_addr.s_addr = htonl(INADDR_ANY);

    if (bind(listen_fd, reinterpret_cast<sockaddr*>(&addr), sizeof(addr)) < 0) {
        close(listen_fd);
        return -1;
    }

    if (listen(listen_fd, 5) < 0) {
        close(listen_fd);
        return -1;
    }

    while (1) {
        sockaddr_in conn_addr{};
        socklen_t conn_len = sizeof(conn_addr);
        int cli_fd = accept(listen_fd, reinterpret_cast<sockaddr*>(&conn_addr), &conn_len);
        if (cli_fd < 0) {
            continue;
        }

        char buf[1024];
        ssize_t n = recv(cli_fd, buf, sizeof(buf) - 1, 0);
        if (n > 0) {
            buf[n] = '\0';
        }

        close(cli_fd);
    }
}

使用注意：
1.sockaddr_in建议初始化用{}，确保没有脏数据
2.每个socket创建成功后后续步骤失败都需要close
3.recv返回值三种情况 >0正常接收 =0正常关闭 <0失败
4.目前是短连接

类型sockaddr：
就是一个通用的地址结构体，经常使用的是类似的结构体sockaddr_in(IPv4)和sockaddr_in6(IPv6)，不是继承关系，但是使用(类型)可以转换
使用到的内部成员变量sin_family sin_port sin_addr.s_addr

socket相关函数：
/**
 * @brief 创建通信端点（套接字）
 * @param domain 协议族/地址族
 *     - AF_INET: IPv4协议
 *     - AF_INET6: IPv6协议  
 * @param type 套接字类型
 *     - SOCK_STREAM: 面向连接的TCP套接字（可靠，双向字节流）
 *     - SOCK_DGRAM: 无连接的UDP套接字（不可靠，数据报文）
 * @param protocol 具体协议类型,通常设为0,由前两个参数自动选择
 * @return 成功返回非负文件描述符，失败返回-1
 */
int socket(int domain, int type, int protocol);
/**
 * @brief 将套接字与本地地址和端口绑定
 * @param sockfd socket()返回的文件描述符
 * @param addr 指向地址结构的通用指针，实际应为sockaddr_in或sockaddr_in6
 * @param addrlen 地址结构的实际长度（字节数）
 * @return 成功返回0，失败返回-1
 */
int bind(int sockfd, const struct sockaddr *addr, socklen_t addrlen);
/**
 * @brief 将主动套接字转换为被动监听套接字
 * @param sockfd 已成功bind()的TCP套接字描述符
 * @param backlog 已完成连接队列的最大长度
 *     - 表示系统可同时接受的连接数
 *     - 典型值：SOMAXCONN（系统最大）或5-10
 * @return 成功返回0，失败返回-1
 */
int listen(int sockfd, int backlog);
